3NFS mount of the storage models dir

apt-get install -y nfs-common
MNT=/var/lib/korvarix-cluster/ollama/models
mkdir -p "$MNT"
# local leftovers (converted boxes only): remove AFTER the mount is down
mountpoint -q "$MNT" && umount "$MNT" && rm -rf "$MNT" && mkdir -p "$MNT"

mount -t nfs -o ro 10.8.0.20:/var/lib/korvarix-cluster/ollama/models "$MNT"
grep -q 'korvarix-cluster/ollama/models' /etc/fstab || \
  echo '10.8.0.20:/var/lib/korvarix-cluster/ollama/models '"$MNT"' nfs ro,defaults,_netdev,noatime 0 0' >> /etc/fstab
mount | grep korvarix           # expect: ...(ro,...
ls "$MNT"/manifests/registry.ollama.ai/oroboros-labs/   # fable manifests visible

ro everywhere: workers never write here (pulls happen on storage). UID parity: id korvarix-ollama must show the same numeric uid as on storage — if not: usermod -u <storage-uid> korvarix-ollama.

4Instance template + enable N instances

N=4
cat > /etc/systemd/system/korvarix-ollama@.service <<'EOF'
[Unit]
Description=korvarix sandboxed Ollama (port %i, 1 request / single slot)
After=network-online.target

[Service]
Type=simple
User=korvarix-ollama
Group=korvarix-ollama
ExecStart=/var/lib/korvarix-cluster/ollama/bin/ollama serve
Environment=OLLAMA_HOST=10.8.0.21:%i
Environment=OLLAMA_MODELS=/var/lib/korvarix-cluster/ollama/models
Environment=HOME=/var/lib/korvarix-cluster/ollama
Environment=OLLAMA_NUM_PARALLEL=1
Environment=OLLAMA_MAX_LOADED_MODELS=1
Environment=OLLAMA_KEEP_ALIVE=-1
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ProtectKernelTunables=true
ProtectKernelModules=true
ProtectControlGroups=true
RestrictSUIDSGID=true
LockPersonality=true
ReadWritePaths=/var/lib/korvarix-cluster/ollama/models /var/lib/korvarix-cluster/ollama
LimitMEMLOCK=64M
LimitNOFILE=65536
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF
systemd-analyze verify /etc/systemd/system/korvarix-ollama@.service   # silence = OK
systemctl daemon-reload

BASE=11430
for i in $(seq 0 $((N-1))); do
  p=$((BASE+i))
  # free the port first (idempotent)
  pid=$(ss -tlnpH "sport = :$p" | grep -oP 'pid=\K[0-9]+' | head -1)
  [ -n "$pid" ] && ! grep -qs "korvarix-ollama@$p" /proc/$pid/cgroup && kill "$pid" && sleep 1
  systemctl enable --now "korvarix-ollama@$p"
done

for i in $(seq 0 $((N-1))); do
  p=$((BASE+i))
  curl -m 3 -s "http://10.8.0.21:$p/api/version" | grep -q version && echo "$p ok" || echo "$p FAIL"
done

One request per instance (NUM_PARALLEL=1, MAX_LOADED=1), KEEP_ALIVE=-1 = models never unload (the NFS model-load stampede never happens again). Port range: 11430 + (N−1).

5Warmup — pre-load the model (permanent)

MODEL=oroboros-labs/claude-fable5:latest
BASE=11430; N=4
for i in $(seq 0 $((N-1))); do
  curl -s "http://10.8.0.21:$((BASE+i))/api/generate" \
    -d "{\"model\":\"$MODEL\",\"prompt\":\"\",\"keep_alive\":-1}" >/dev/null &
done; wait
# first load pulls ~20 GB over NFS per instance — run in a screen/tmux and let it finish
for i in $(seq 0 $((N-1))); do
  printf "%s " "$((BASE+i))"; curl -s "http://10.8.0.21:$((BASE+i))/api/ps" | grep -c claude-fable5
done    # expect N lines of "1"

7OWUI per-request threads (one-time, global)

Admin Panel → Settings → Models → edit the fable5 model → Advanced Params → num_thread = 16. Per-request, applies live — this is the "cores per task" lever. With N=4 × 16 threads a worker uses all 64 cores at full concurrency.
